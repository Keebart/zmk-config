  # Replace only this build's stock renderer. Never modify the upstream source or other targets.
  function(keebart_renderer)
    get_target_property(sources app SOURCES)
    if(CONFIG_KEEBART_STUDIO)
      list(FILTER sources EXCLUDE REGEX "(^|/)hid_listener\\.c$")
    endif()
    list(FILTER sources EXCLUDE REGEX "(^|/)rgb_underglow\\.c$")
    set_property(TARGET app PROPERTY SOURCES ${sources})
  endfunction()
  cmake_language(DEFER DIRECTORY ${APPLICATION_SOURCE_DIR} CALL keebart_renderer)

# Scope the correction to the Studio build, leaving upstream/default firmware alone.
target_include_directories(zephyr PRIVATE "${CMAKE_CURRENT_LIST_DIR}/..")
set(ks_nvs_dir "${ZEPHYR_BASE}/subsys/fs/nvs")
set(ks_nvs_source "${CMAKE_CURRENT_BINARY_DIR}/nvs.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${ks_nvs_dir}/nvs.c" "${CMAKE_CURRENT_LIST_DIR}/nvs_recovery.py")
execute_process(
  COMMAND ${PYTHON_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/nvs_recovery.py"
          "${ks_nvs_dir}/nvs.c" "${ks_nvs_source}"
  RESULT_VARIABLE ks_nvs_result
)
if(NOT ks_nvs_result EQUAL 0)
  message(FATAL_ERROR "Studio NVS recovery fix could not be applied")
endif()
# nvs.c includes this private header relative to its source directory.
configure_file("${ks_nvs_dir}/nvs_priv.h" "${CMAKE_CURRENT_BINARY_DIR}/nvs_priv.h" COPYONLY)
set_property(GLOBAL PROPERTY KEEBART_NVS_SOURCE "${ks_nvs_source}")
function(keebart_nvs_recovery)
  get_target_property(sources zephyr SOURCES)
  set(originals ${sources})
  list(FILTER originals INCLUDE REGEX "(^|/)subsys/fs/nvs/nvs\\.c$")
  list(LENGTH originals count)
  if(NOT count EQUAL 1)
    message(FATAL_ERROR "Studio expects exactly one pinned NVS source")
  endif()
  list(REMOVE_ITEM sources ${originals})
  get_property(replacement GLOBAL PROPERTY KEEBART_NVS_SOURCE)
  list(APPEND sources "${replacement}")
  set_property(TARGET zephyr PROPERTY SOURCES ${sources})
endfunction()
cmake_language(DEFER DIRECTORY ${APPLICATION_SOURCE_DIR} CALL keebart_nvs_recovery)

# Studio mouse keys expose the same eight buttons as wired boards.
if(CONFIG_KEEBART_STUDIO)
  set(pointer_header "${APPLICATION_SOURCE_DIR}/include/zmk/hid.h")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${pointer_header}")
  file(READ "${pointer_header}" pointer_hid)
  string(REPLACE "#define ZMK_HID_MOUSE_NUM_BUTTONS 0x05" "#define ZMK_HID_MOUSE_NUM_BUTTONS 0x08" pointer_hid "${pointer_hid}")
  string(REPLACE "HID_REPORT_COUNT(0x5)," "HID_REPORT_COUNT(0x8)," pointer_hid "${pointer_hid}")
  string(REPLACE "    // Constant padding for the last 3 bits.\n    HID_REPORT_SIZE(0x03),\n    HID_REPORT_COUNT(0x01),\n    HID_INPUT(ZMK_HID_MAIN_VAL_CONST | ZMK_HID_MAIN_VAL_VAR | ZMK_HID_MAIN_VAL_ABS),\n" "" pointer_hid "${pointer_hid}")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/pointer_include/zmk")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pointer_include/zmk/hid.h" "${pointer_hid}")
  target_include_directories(zephyr_interface BEFORE INTERFACE "${CMAKE_CURRENT_BINARY_DIR}/pointer_include")
endif()
